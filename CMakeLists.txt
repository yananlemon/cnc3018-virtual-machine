cmake_minimum_required(VERSION 3.21)
project(cnc3018_virtual_machine VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CNC_BUILD_DESKTOP "Build Qt desktop application" ON)
function(cnc_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
add_library(cnc_controller src/core/controller.cpp src/core/stock.cpp)
target_include_directories(cnc_controller PUBLIC src)
target_compile_features(cnc_controller PUBLIC cxx_std_20)
cnc_warnings(cnc_controller)
add_library(cnc_transport src/transport/win_serial.cpp)
target_compile_features(cnc_transport PUBLIC cxx_std_20)
target_include_directories(cnc_transport PUBLIC src)
cnc_warnings(cnc_transport)
include(CTest)
if(BUILD_TESTING)
  add_executable(stock_tests tests/stock_tests.cpp)
  target_link_libraries(stock_tests PRIVATE cnc_controller)
  cnc_warnings(stock_tests)
  add_test(NAME material_removal_contract COMMAND stock_tests)
  add_executable(controller_tests tests/controller_tests.cpp)
  target_link_libraries(controller_tests PRIVATE cnc_controller)
  cnc_warnings(controller_tests)
  foreach(suite protocol coordinates streamer numeric origin)
    add_test(NAME ${suite}_contract COMMAND controller_tests ${suite})
  endforeach()
  add_test(NAME demo_contract COMMAND controller_tests demo "${CMAKE_CURRENT_SOURCE_DIR}/examples/3018-demo.nc")
  add_executable(transport_tests tests/transport_tests.cpp)
  target_link_libraries(transport_tests PRIVATE cnc_transport)
  cnc_warnings(transport_tests)
  add_test(NAME transport_lifecycle COMMAND transport_tests)
endif()
if(CNC_BUILD_DESKTOP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL)
  add_executable(cnc3018_vm WIN32 src/ui/main.cpp src/ui/setup_document.cpp src/ui/program.cpp src/ui/machine_view.cpp src/ui/mesh.cpp)
  target_link_libraries(cnc3018_vm PRIVATE cnc_controller cnc_transport Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL)
  cnc_warnings(cnc3018_vm)
  if(BUILD_TESTING)
    add_executable(setup_tests tests/setup_tests.cpp src/ui/setup_document.cpp)
    target_link_libraries(setup_tests PRIVATE cnc_controller Qt6::Core)
    cnc_warnings(setup_tests)
    add_test(NAME setup_fixture_contract COMMAND setup_tests)
    add_executable(program_tests tests/program_tests.cpp src/ui/program.cpp)
    target_link_libraries(program_tests PRIVATE cnc_controller Qt6::Core)
    cnc_warnings(program_tests)
    add_test(NAME gcode_import_contract COMMAND program_tests)
    add_executable(mesh_tests tests/mesh_tests.cpp src/ui/mesh.cpp)
    target_include_directories(mesh_tests PRIVATE src)
    target_link_libraries(mesh_tests PRIVATE Qt6::Gui)
    cnc_warnings(mesh_tests)
    add_test(NAME mechanical_mesh_contract COMMAND mesh_tests)
  endif()
  if(MINGW)
    target_link_options(cnc3018_vm PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
